  ENDMETHOD.

"--------------------------------------------------------------------
" Local class: File system utilities
"--------------------------------------------------------------------
CLASS lcl_file_utils DEFINITION FINAL.
  PUBLIC SECTION.

    "! Reads all .zip file names from a LOCAL (presentation server) directory or single file.
    "! @parameter iv_path    | Local directory path or file
    "! @parameter et_files   | List of full paths found
    CLASS-METHODS get_local_zip_files
      IMPORTING iv_path          TYPE string
      EXPORTING et_files         TYPE string_table
      RAISING   cx_parameter_invalid.

    "! Reads all .zip file names from a SERVER (application server) directory.
    "! @parameter iv_path    | Server directory path
    "! @parameter et_files   | List of full paths found
    CLASS-METHODS get_server_zip_files
      IMPORTING iv_path          TYPE string
      EXPORTING et_files         TYPE string_table
      RAISING   cx_parameter_invalid.

    "! Reads raw binary content of a file from the PRESENTATION server.
    "! @parameter iv_filepath  | Full file path
    "! @parameter ev_xstring   | Binary content
    CLASS-METHODS read_local_file
      IMPORTING iv_filepath      TYPE string
      EXPORTING ev_xstring       TYPE xstring
      RAISING   cx_parameter_invalid.

    "! Reads raw binary content of a file from the APPLICATION server.
    "! @parameter iv_filepath  | Full file path
    "! @parameter ev_xstring   | Binary content
    CLASS-METHODS read_server_file
      IMPORTING iv_filepath      TYPE string
      EXPORTING ev_xstring       TYPE xstring
      RAISING   cx_parameter_invalid.

    "! Extracts pure file name from a full path string.
    "! @parameter iv_fullpath | Full path
    "! @returning rv_name     | File name only
    CLASS-METHODS get_filename_from_path
      IMPORTING iv_fullpath      TYPE string
      RETURNING VALUE(rv_name)   TYPE string.

ENDCLASS.

CLASS lcl_file_utils IMPLEMENTATION.

  METHOD get_local_zip_files.
    CLEAR et_files.
    DATA lt_raw  TYPE filetable.
    DATA lv_rc   TYPE i.
    DATA lv_is_file TYPE abap_bool VALUE abap_false.

    "-- Check if iv_path is a file (ends with .zip/.ZIP)
    IF iv_path CS '.zip' OR iv_path CS '.ZIP'.
      lv_is_file = abap_true.
    ENDIF.

    IF lv_is_file = abap_true.
      "-- Single file mode
      APPEND iv_path TO et_files.
      RETURN.
    ENDIF.

    "-- Directory mode
    cl_gui_frontend_services=>directory_list_files(
      EXPORTING
        directory  = iv_path
        filter     = '*.zip'
      CHANGING
        file_table = lt_raw
        count      = lv_rc
      EXCEPTIONS
        cntl_error           = 1
        wrong_parameter      = 2
        error_no_gui         = 3
        not_supported_by_gui = 4
        OTHERS               = 5 ).

    IF sy-subrc <> 0.
      RAISE EXCEPTION TYPE cx_parameter_invalid
        EXPORTING
          parameter = CONV #( |Directory not found or error: { iv_path }| ).
    ENDIF.

    LOOP AT lt_raw INTO DATA(ls_file).
      DATA(lv_full) = |{ iv_path }{ ls_file-filename }|.
      IF NOT ( lv_full CS '.zip' OR lv_full CS '.ZIP' ).
        CONTINUE.
      ENDIF.
      APPEND lv_full TO et_files.
    ENDLOOP.
  ENDMETHOD.

  METHOD get_server_zip_files.
    CLEAR et_files.
    DATA(lv_base_path) = iv_path.

    IF lv_base_path IS NOT INITIAL.
      DATA(lv_base_last) = substring( val = lv_base_path off = strlen( lv_base_path ) - 1 len = 1 ).
      IF lv_base_last <> '/' AND lv_base_last <> '\'.
        lv_base_path = |{ lv_base_path }/|.
      ENDIF.
    ENDIF.

    "-- List files on application server using EPS2_GET_DIRECTORY_LISTING
    DATA lt_dir  TYPE TABLE OF eps2fili.
    DATA lv_dir_name      TYPE eps2filnam.
    DATA lv_real_dir_name TYPE epsf-epsdirnam.
    DATA lv_file_counter  TYPE epsf-epsfilsiz.
    DATA lv_error_counter TYPE epsf-epsfilsiz.
    lv_dir_name = CONV eps2filnam( lv_base_path ).

    TRY.
        CALL FUNCTION 'EPS2_GET_DIRECTORY_LISTING'
          EXPORTING
            iv_dir_name        = lv_dir_name
            file_mask          = '*.zip*'
          IMPORTING
            dir_name           = lv_real_dir_name
            file_counter       = lv_file_counter
            error_counter      = lv_error_counter
          TABLES
            dir_list           = lt_dir
          EXCEPTIONS
            invalid_eps_subdir = 1
            sapgparam_failed   = 2
            build_directory_failed = 3
            no_authorization   = 4
            read_directory_failed = 5
            too_many_read_errors = 6
            empty_directory_list = 7
            OTHERS             = 8.
      CATCH cx_sy_dyn_call_illegal_type.
        RAISE EXCEPTION TYPE cx_parameter_invalid
          EXPORTING
            parameter = CONV #( |Invalid directory type/value: { iv_path }| ).
    ENDTRY.

    IF sy-subrc = 7.
      RETURN.
    ENDIF.

    IF sy-subrc <> 0.
      RAISE EXCEPTION TYPE cx_parameter_invalid
        EXPORTING
          parameter = CONV #( |Server directory error: { iv_path }| ).
    ENDIF.

    LOOP AT lt_dir INTO DATA(ls_dir).
      DATA(lv_name) = CONV string( ls_dir-name ).
      DATA(lv_name_len) = strlen( lv_name ).
      DATA lv_ext TYPE string.
      IF lv_name_len >= 4.
        lv_ext = substring( val = lv_name off = lv_name_len - 4 len = 4 ).
      ENDIF.
      IF lv_name_len >= 4
         AND ( lv_ext = '.zip'
               OR lv_ext = '.ZIP' ).
        DATA(lv_full_path) = |{ lv_base_path }{ lv_name }|.
        APPEND lv_full_path TO et_files.
      ENDIF.
    ENDLOOP.
  ENDMETHOD.

  METHOD read_local_file.
    CLEAR ev_xstring.
    DATA lt_hex TYPE TABLE OF x255.
    DATA lv_length TYPE i.

    cl_gui_frontend_services=>gui_upload(
      EXPORTING
        filename   = iv_filepath
        filetype   = 'BIN'
      IMPORTING
        filelength = lv_length
      CHANGING
        data_tab   = lt_hex
      EXCEPTIONS
        file_open_error     = 1
        file_read_error     = 2
        no_batch            = 3
        gui_refuse_filetransfer = 4
        invalid_type        = 5
        no_authority        = 6
        OTHERS              = 7 ).

    IF sy-subrc <> 0.
      RAISE EXCEPTION TYPE cx_parameter_invalid
        EXPORTING
          parameter = CONV #( |Cannot read local file: { iv_filepath }| ).
    ENDIF.

    "-- Concatenate binary chunks into xstring
    DATA lv_xstr TYPE xstring.
    LOOP AT lt_hex INTO DATA(lv_chunk).
      lv_xstr = lv_xstr && lv_chunk.
    ENDLOOP.
    "-- Trim to actual file length
    ev_xstring = lv_xstr(lv_length).
  ENDMETHOD.

  METHOD read_server_file.
    CLEAR ev_xstring.

    "-- Open file on application server
    OPEN DATASET iv_filepath FOR INPUT IN BINARY MODE.
    IF sy-subrc <> 0.
      RAISE EXCEPTION TYPE cx_parameter_invalid
        EXPORTING
          parameter = CONV #( |Cannot open server file: { iv_filepath }| ).
    ENDIF.

    "-- Preferred path: read full binary payload directly into xstring
    READ DATASET iv_filepath INTO ev_xstring.
    IF sy-subrc = 0 AND ev_xstring IS NOT INITIAL.
      CLOSE DATASET iv_filepath.
      RETURN.
    ENDIF.

    "-- Fallback path for kernels/filesystems not returning full xstring in one read
    CLOSE DATASET iv_filepath.
    OPEN DATASET iv_filepath FOR INPUT IN BINARY MODE.
    IF sy-subrc <> 0.
      RAISE EXCEPTION TYPE cx_parameter_invalid
        EXPORTING
          parameter = CONV #( |Cannot reopen server file: { iv_filepath }| ).
    ENDIF.

    DATA lv_chunk          TYPE x LENGTH 1024.
    DATA lv_actual_length  TYPE i.

    "-- Read full file
    DATA lv_raw TYPE xstring.
    DO.
      CLEAR lv_chunk.
      READ DATASET iv_filepath INTO lv_chunk MAXIMUM LENGTH 1024 ACTUAL LENGTH lv_actual_length.
      IF sy-subrc <> 0.
        EXIT.
      ENDIF.
      IF lv_actual_length > 0.
        lv_raw = lv_raw && lv_chunk(lv_actual_length).
      ENDIF.
    ENDDO.
    CLOSE DATASET iv_filepath.
    ev_xstring = lv_raw.
  ENDMETHOD.

  METHOD get_filename_from_path.
    "-- Works for both Unix '/' and Windows '\' separators
    DATA lv_sep_pos TYPE i VALUE 0.
    DATA(lv_len)        = strlen( iv_fullpath ).
    DATA lv_i TYPE i.

    DO lv_len TIMES.
      lv_i = lv_len - sy-index.
      DATA(lv_char) = substring( val = iv_fullpath off = lv_i len = 1 ).
      IF lv_char = '/' OR lv_char = '\'.
        lv_sep_pos = lv_i + 1.
        EXIT.
      ENDIF.
    ENDDO.

    rv_name = substring( val = iv_fullpath off = lv_sep_pos ).
  ENDMETHOD.

ENDCLASS.

"--------------------------------------------------------------------
" Local class: ZIP integrity verifier
"--------------------------------------------------------------------
CLASS lcl_zip_verifier DEFINITION FINAL.
  PUBLIC SECTION.

    "! Verifies a ZIP xstring:
    "!   1. ZIP structure and EOCD consistency
    "!   2. At least one PDF entry in central directory
    "!   3. Extraction and basic integrity check of PDF payload
    "! @parameter iv_xstring    | Raw ZIP binary content
    "! @parameter ev_ok         | ABAP_TRUE if integrity ok
    "! @parameter ev_message    | Descriptive result message
    CLASS-METHODS verify_zip_integrity
      IMPORTING iv_xstring    TYPE xstring
                iv_strict     TYPE abap_bool DEFAULT abap_true
      EXPORTING ev_ok         TYPE abap_bool
                ev_message    TYPE string
               ev_warning    TYPE string
                ev_eocd_pos   TYPE i
                ev_zip_size   TYPE i
                ev_trailing_b TYPE i.

  PRIVATE SECTION.

    TYPES:
      BEGIN OF ty_zip_entry,
        name        TYPE string,
        name_xstr   TYPE xstring,
          zip_index   TYPE i,
          is_pdf      TYPE abap_bool,
        gp_flags    TYPE i,
        method      TYPE i,
        crc32       TYPE i,
        comp_size   TYPE i,
        uncomp_size TYPE i,
        local_off   TYPE i,
      END OF ty_zip_entry,
      ty_t_zip_entry TYPE STANDARD TABLE OF ty_zip_entry WITH EMPTY KEY.

    CONSTANTS:
      c_zip_local_sig   TYPE x LENGTH 4 VALUE '504B0304', "< PK\x03\x04
      c_zip_eocd_sig    TYPE x LENGTH 4 VALUE '504B0506', "< PK\x05\x06
      c_zip_cdir_sig    TYPE x LENGTH 4 VALUE '504B0102', "< PK\x01\x02
      c_zip_empty_sig   TYPE x LENGTH 4 VALUE '504B0506', "< empty archive starts with EOCD
      c_pdf_magic       TYPE x LENGTH 4 VALUE '25504446', "< %PDF
      c_pdf_ext_upper   TYPE x LENGTH 4 VALUE '2E504446', "< .PDF
      c_pdf_ext_lower   TYPE x LENGTH 4 VALUE '2E706466', "< .pdf
      c_pdf_eof_upper   TYPE x LENGTH 5 VALUE '2525454F46', "< %%EOF
      c_pdf_eof_lower   TYPE x LENGTH 5 VALUE '2525656F66'. "< %%eof

    "! Searches for a byte pattern in an xstring.
    "! @parameter iv_haystack | Binary data to search in
    "! @parameter iv_needle   | Pattern to find
    "! @returning rv_pos      | 0-based position or -1 if not found
    CLASS-METHODS find_pattern
      IMPORTING iv_haystack    TYPE xstring
                iv_needle      TYPE xstring
      RETURNING VALUE(rv_pos)  TYPE i.

    "! Finds EOCD in ZIP tail and validates comment length.
    CLASS-METHODS find_eocd_position
      IMPORTING iv_zip_data     TYPE xstring
      RETURNING VALUE(rv_pos)   TYPE i.

    "! Extracts PDF entry names from central directory.
    CLASS-METHODS get_pdf_entries_from_cdir
      IMPORTING iv_zip_data      TYPE xstring
                iv_eocd_pos      TYPE i
      EXPORTING et_pdf_entries   TYPE string_table
                ev_total_entries TYPE i
                et_entries       TYPE ty_t_zip_entry
      RAISING   cx_parameter_invalid.

    "! Validates consistency between central directory entry and local header.
    CLASS-METHODS validate_entry_consistency
      IMPORTING iv_zip_data      TYPE xstring
                is_entry         TYPE ty_zip_entry
      EXPORTING ev_ok            TYPE abap_bool
                ev_message       TYPE string.

    "! Performs lightweight PDF validation on extracted payload.
    CLASS-METHODS verify_pdf_payload
      IMPORTING iv_pdf_data       TYPE xstring
      EXPORTING ev_ok             TYPE abap_bool
                ev_message        TYPE string.

    "! Reads a 2-byte little-endian unsigned integer from xstring at offset.
    CLASS-METHODS read_uint16_le
      IMPORTING iv_xstring  TYPE xstring
                iv_offset   TYPE i
      RETURNING VALUE(rv)   TYPE i.

    "! Reads a 4-byte little-endian unsigned integer from xstring at offset.
    CLASS-METHODS read_uint32_le
      IMPORTING iv_xstring  TYPE xstring
                iv_offset   TYPE i
      RETURNING VALUE(rv)   TYPE i.

ENDCLASS.

CLASS lcl_zip_verifier IMPLEMENTATION.

  METHOD verify_zip_integrity.
    ev_ok      = abap_false.
    ev_message = 'Unknown error'.
    ev_warning = ''.
    ev_eocd_pos   = -1.
    ev_trailing_b = -1.
    DATA(lv_zip_data) = iv_xstring.

    DATA(lv_size) = xstrlen( lv_zip_data ).
    ev_zip_size = lv_size.

    "-- 1. Minimum size check (at least local file header: 30 bytes + EOCD: 22 bytes)
    IF lv_size < 22.
      ev_message = 'File too small to be a valid ZIP archive'.
      RETURN.
    ENDIF.

    "-- 2. ZIP start signature: local header or empty archive (EOCD)
    DATA(lv_sig) = lv_zip_data(4).
    IF lv_sig <> c_zip_local_sig AND lv_sig <> c_zip_empty_sig.
      ev_message = 'Invalid ZIP signature - file is corrupted or not a ZIP'.
      RETURN.
    ENDIF.

    "-- 3. Search End-of-Central-Directory (EOCD) in ZIP tail
    DATA(lv_eocd_pos) = find_eocd_position( lv_zip_data ).
    ev_eocd_pos = lv_eocd_pos.

    IF lv_eocd_pos < 0.
      " In strict mode, missing EOCD is treated as structural corruption.
      IF iv_strict = abap_true.
        ev_message = 'End-of-Central-Directory record not found - ZIP is corrupted in strict mode'.
        RETURN.
      ENDIF.

      " Non-strict fallback for non-standard archives where EOCD signature is not found.
      DATA(lo_zip_fallback) = NEW cl_abap_zip( ).
      DATA(lv_pdf_hint_upper) = find_pattern(
        iv_haystack = lv_zip_data
        iv_needle   = CONV xstring( c_pdf_ext_upper ) ).
      DATA(lv_pdf_hint_lower) = find_pattern(
        iv_haystack = lv_zip_data
        iv_needle   = CONV xstring( c_pdf_ext_lower ) ).

      CALL METHOD lo_zip_fallback->load
        EXPORTING
          zip = lv_zip_data
        EXCEPTIONS
          zip_parse_error = 1
          OTHERS          = 2.

      IF sy-subrc <> 0.
        ev_message = 'End-of-Central-Directory record not found - ZIP may be truncated'.
        RETURN.
      ENDIF.

      IF lv_pdf_hint_upper < 0 AND lv_pdf_hint_lower < 0.
        ev_message = 'ZIP loaded by fallback but no PDF entry marker found (EOCD signature not found)'.
        RETURN.
      ENDIF.

      ev_ok = abap_true.
      ev_message = 'ZIP loaded by fallback (EOCD signature not found) and PDF marker found'.
      RETURN.
    ENDIF.

    DATA(lv_eocd_comment_len) = read_uint16_le( iv_xstring = lv_zip_data iv_offset = lv_eocd_pos + 20 ).
    DATA(lv_eocd_end) = lv_eocd_pos + 22 + lv_eocd_comment_len.
    ev_trailing_b = lv_size - lv_eocd_end.
    DATA(lv_eocd_has_trailing) = COND abap_bool( WHEN lv_eocd_end < lv_size THEN abap_true ELSE abap_false ).
    DATA(lv_eocd_note) = COND string(
      WHEN lv_eocd_has_trailing = abap_true THEN ' EOCD found with trailing bytes ignored.'
      ELSE '' ).

    "-- 4. Parse central directory and collect PDF entries
    DATA lt_pdf_entries TYPE string_table.
    DATA lv_entry_count TYPE i.
    DATA lt_entries TYPE ty_t_zip_entry.
    DATA(lv_consistency_note) = VALUE string( ).

    TRY.
        get_pdf_entries_from_cdir(
          EXPORTING iv_zip_data      = lv_zip_data
                    iv_eocd_pos      = lv_eocd_pos
          IMPORTING et_pdf_entries   = lt_pdf_entries
                    ev_total_entries = lv_entry_count
                    et_entries       = lt_entries ).
      CATCH cx_parameter_invalid INTO DATA(lx_cdir).
        ev_message = lx_cdir->get_text( ).
        RETURN.
    ENDTRY.

    LOOP AT lt_entries INTO DATA(ls_entry).
      DATA lv_cons_ok TYPE abap_bool.
      DATA lv_cons_msg TYPE string.
      validate_entry_consistency(
        EXPORTING iv_zip_data = lv_zip_data
                  is_entry    = ls_entry
        IMPORTING ev_ok       = lv_cons_ok
                  ev_message  = lv_cons_msg ).
      IF lv_cons_ok = abap_false.
        IF lv_consistency_note IS INITIAL.
          lv_consistency_note = | Entry consistency warning for '{ ls_entry-name }': { lv_cons_msg }.|.
        ENDIF.
      ENDIF.
    ENDLOOP.

    IF lv_entry_count = 0.
      ev_message = 'ZIP archive is empty - no entries found'.
      RETURN.
    ENDIF.

    IF lt_pdf_entries IS INITIAL.
      ev_message = |ZIP is structurally valid but contains no PDF file ({ lv_entry_count } entries).{ lv_eocd_note }|.
      RETURN.
    ENDIF.

    IF iv_strict = abap_false.
      ev_ok      = abap_true.
      ev_message = |ZIP structure OK and PDF entry present ({ lines( lt_pdf_entries ) } PDF).{ lv_eocd_note }|.
      ev_warning = lv_consistency_note.
      RETURN.
    ENDIF.

    "-- 5. Extract each PDF entry and validate payload
    DATA(lo_zip) = NEW cl_abap_zip( ).
    CALL METHOD lo_zip->load
      EXPORTING
        zip = lv_zip_data
      EXCEPTIONS
        zip_parse_error = 1
        OTHERS          = 2.

    IF sy-subrc <> 0.
      ev_message = 'ZIP load failed (ZIP_PARSE_ERROR)'.
      RETURN.
    ENDIF.

    DATA(lv_pdf_names) = VALUE string( ).

    LOOP AT lt_entries INTO DATA(ls_pdf_entry) WHERE is_pdf = abap_true.
      DATA lv_pdf_data TYPE xstring.
      CALL METHOD lo_zip->get
        EXPORTING
          index   = ls_pdf_entry-zip_index
        IMPORTING
          content = lv_pdf_data
        EXCEPTIONS
          zip_index_error = 1
          OTHERS          = 2.

      IF sy-subrc <> 0.
        ev_message = |Cannot extract PDF entry #{ ls_pdf_entry-zip_index } from ZIP (ZIP_INDEX_ERROR)|.
        RETURN.
      ENDIF.

      DATA(lv_pdf_len) = xstrlen( lv_pdf_data ).
      IF ls_pdf_entry-uncomp_size > 0
         AND ls_pdf_entry-uncomp_size <= lv_size
         AND lv_pdf_len <> ls_pdf_entry-uncomp_size.
        IF lv_consistency_note IS INITIAL.
          lv_consistency_note = | PDF size warning for entry #{ ls_pdf_entry-zip_index }: expected { ls_pdf_entry-uncomp_size }, got { lv_pdf_len }.|.
        ENDIF.
      ENDIF.

      DATA(lv_pdf_name) = COND string(
        WHEN ls_pdf_entry-name IS INITIAL THEN |ENTRY#{ ls_pdf_entry-zip_index }|
        ELSE ls_pdf_entry-name ).

      DATA lv_pdf_ok TYPE abap_bool.
      DATA lv_pdf_msg TYPE string.

      verify_pdf_payload(
        EXPORTING iv_pdf_data = lv_pdf_data
        IMPORTING ev_ok       = lv_pdf_ok
                  ev_message  = lv_pdf_msg ).

      IF lv_pdf_ok = abap_false.
        ev_message = |PDF '{ lv_pdf_name }' is corrupted: { lv_pdf_msg }|.
        RETURN.
      ENDIF.

      lv_pdf_names = |{ lv_pdf_names }{ lv_pdf_name } |.
    ENDLOOP.

    "-- 6. Final verdict
    ev_ok      = abap_true.
    ev_message = |ZIP and PDF integrity OK. PDF found: { lv_pdf_names }.|.
    ev_warning = |{ lv_eocd_note }{ lv_consistency_note }|.
  ENDMETHOD.

  METHOD find_eocd_position.
    rv_pos = -1.

    DATA(lv_size) = xstrlen( iv_zip_data ).
    IF lv_size < 22.
      RETURN.
    ENDIF.

    " First pass: standard ZIP window (22 fixed + 65,535 comment)
    DATA(lv_search_start) = COND i(
      WHEN lv_size > 65557 THEN lv_size - 65557
      ELSE 0 ).

    DATA lv_i TYPE i.
    lv_i = lv_size - 22.

    WHILE lv_i >= lv_search_start.
      IF iv_zip_data+lv_i(4) = c_zip_eocd_sig.
        DATA(lv_comment_len) = read_uint16_le( iv_xstring = iv_zip_data iv_offset = lv_i + 20 ).
        " Some producer chains may append trailing bytes after EOCD.
        " Accept EOCD if the declared comment length is consistent and in range.
        IF lv_i + 22 + lv_comment_len <= lv_size.
          rv_pos = lv_i.
          RETURN.
        ENDIF.
      ENDIF.
      lv_i = lv_i - 1.
    ENDWHILE.

    " Fallback: scan full file for non-standard archives with very long trailing bytes.
    lv_i = lv_size - 22.
    WHILE lv_i >= 0.
      IF iv_zip_data+lv_i(4) = c_zip_eocd_sig.
        lv_comment_len = read_uint16_le( iv_xstring = iv_zip_data iv_offset = lv_i + 20 ).
        IF lv_i + 22 + lv_comment_len <= lv_size.
          rv_pos = lv_i.
          RETURN.
        ENDIF.
      ENDIF.
      lv_i = lv_i - 1.
    ENDWHILE.
  ENDMETHOD.

  METHOD get_pdf_entries_from_cdir.
    CLEAR et_pdf_entries.
    ev_total_entries = 0.
    CLEAR et_entries.

    DATA(lv_size) = xstrlen( iv_zip_data ).
    IF iv_eocd_pos < 0 OR iv_eocd_pos + 22 > lv_size.
      RAISE EXCEPTION TYPE cx_parameter_invalid
        EXPORTING parameter = 'EOCD position invalid'.
    ENDIF.

    DATA(lv_entry_count) = read_uint16_le( iv_xstring = iv_zip_data iv_offset = iv_eocd_pos + 10 ).
    DATA(lv_cdir_size)   = read_uint32_le( iv_xstring = iv_zip_data iv_offset = iv_eocd_pos + 12 ).
    DATA(lv_cdir_offset) = read_uint32_le( iv_xstring = iv_zip_data iv_offset = iv_eocd_pos + 16 ).

    ev_total_entries = lv_entry_count.

    IF lv_entry_count = 0.
      RETURN.
    ENDIF.

    IF lv_cdir_offset < 0 OR lv_cdir_offset >= lv_size.
      RAISE EXCEPTION TYPE cx_parameter_invalid
        EXPORTING parameter = 'Central directory offset out of range'.
    ENDIF.

    IF lv_cdir_size > 0 AND lv_cdir_offset + lv_cdir_size > lv_size.
      RAISE EXCEPTION TYPE cx_parameter_invalid
        EXPORTING parameter = 'Central directory size out of range'.
    ENDIF.

    DATA lv_cur_offset TYPE i.
    DATA lv_entry_idx  TYPE i VALUE 0.

    lv_cur_offset = lv_cdir_offset.

    WHILE lv_entry_idx < lv_entry_count.
      IF lv_cur_offset + 46 > lv_size.
        RAISE EXCEPTION TYPE cx_parameter_invalid
          EXPORTING parameter = 'Central directory entry exceeds ZIP size'.
      ENDIF.

      IF iv_zip_data+lv_cur_offset(4) <> c_zip_cdir_sig.
        RAISE EXCEPTION TYPE cx_parameter_invalid
          EXPORTING parameter = |Central directory entry #{ lv_entry_idx + 1 } has invalid signature|.
      ENDIF.

      DATA(lv_fname_len)   = read_uint16_le( iv_xstring = iv_zip_data iv_offset = lv_cur_offset + 28 ).
      DATA(lv_extra_len)   = read_uint16_le( iv_xstring = iv_zip_data iv_offset = lv_cur_offset + 30 ).
      DATA(lv_comment_len) = read_uint16_le( iv_xstring = iv_zip_data iv_offset = lv_cur_offset + 32 ).
      DATA(lv_gp_flags)    = read_uint16_le( iv_xstring = iv_zip_data iv_offset = lv_cur_offset + 8 ).
      DATA(lv_method)      = read_uint16_le( iv_xstring = iv_zip_data iv_offset = lv_cur_offset + 10 ).
      DATA(lv_crc32)       = read_uint32_le( iv_xstring = iv_zip_data iv_offset = lv_cur_offset + 16 ).
      DATA(lv_comp_size)   = read_uint32_le( iv_xstring = iv_zip_data iv_offset = lv_cur_offset + 20 ).
      DATA(lv_uncomp_size) = read_uint32_le( iv_xstring = iv_zip_data iv_offset = lv_cur_offset + 24 ).
      DATA(lv_local_off)   = read_uint32_le( iv_xstring = iv_zip_data iv_offset = lv_cur_offset + 42 ).
      DATA(lv_name_offset) = lv_cur_offset + 46.

      IF lv_fname_len > 0 AND lv_name_offset + lv_fname_len <= lv_size.
        DATA(lv_fname_xstr) = iv_zip_data+lv_name_offset(lv_fname_len).
        DATA(lv_name_string) = CONV string( lv_fname_xstr ).

        DATA(lv_pdf_pos_upper) = find_pattern(
          iv_haystack = lv_fname_xstr
          iv_needle   = CONV xstring( c_pdf_ext_upper ) ).
        DATA(lv_pdf_pos_lower) = find_pattern(
          iv_haystack = lv_fname_xstr
          iv_needle   = CONV xstring( c_pdf_ext_lower ) ).
        DATA(lv_is_pdf) = COND abap_bool(
          WHEN lv_pdf_pos_upper >= 0 OR lv_pdf_pos_lower >= 0 THEN abap_true
          ELSE abap_false ).

        APPEND VALUE ty_zip_entry(
          name        = lv_name_string
          name_xstr   = lv_fname_xstr
          zip_index   = lv_entry_idx + 1
          is_pdf      = lv_is_pdf
          gp_flags    = lv_gp_flags
          method      = lv_method
          crc32       = lv_crc32
          comp_size   = lv_comp_size
          uncomp_size = lv_uncomp_size
          local_off   = lv_local_off ) TO et_entries.

        IF lv_is_pdf = abap_true.
          APPEND lv_name_string TO et_pdf_entries.
        ENDIF.
      ENDIF.

      lv_cur_offset = lv_cur_offset + 46 + lv_fname_len + lv_extra_len + lv_comment_len.
      lv_entry_idx  = lv_entry_idx + 1.
    ENDWHILE.
  ENDMETHOD.

  METHOD validate_entry_consistency.
    ev_ok = abap_false.
    ev_message = 'Unknown entry consistency error'.

    DATA(lv_size) = xstrlen( iv_zip_data ).

    IF is_entry-local_off < 0 OR is_entry-local_off + 30 > lv_size.
      ev_message = 'Local header offset out of range'.
      RETURN.
    ENDIF.

    IF iv_zip_data+is_entry-local_off(4) <> c_zip_local_sig.
      ev_message = 'Invalid local header signature'.
      RETURN.
    ENDIF.

    DATA(lv_lh_flags)      = read_uint16_le( iv_xstring = iv_zip_data iv_offset = is_entry-local_off + 6 ).
    DATA(lv_lh_method)     = read_uint16_le( iv_xstring = iv_zip_data iv_offset = is_entry-local_off + 8 ).
    DATA(lv_lh_crc32)      = read_uint32_le( iv_xstring = iv_zip_data iv_offset = is_entry-local_off + 14 ).
    DATA(lv_lh_comp_size)  = read_uint32_le( iv_xstring = iv_zip_data iv_offset = is_entry-local_off + 18 ).
    DATA(lv_lh_uncomp_size) = read_uint32_le( iv_xstring = iv_zip_data iv_offset = is_entry-local_off + 22 ).
    DATA(lv_lh_fname_len)  = read_uint16_le( iv_xstring = iv_zip_data iv_offset = is_entry-local_off + 26 ).
    DATA(lv_lh_extra_len)  = read_uint16_le( iv_xstring = iv_zip_data iv_offset = is_entry-local_off + 28 ).

    DATA(lv_name_off) = is_entry-local_off + 30.
    IF lv_name_off + lv_lh_fname_len > lv_size.
      ev_message = 'Local header file name exceeds ZIP size'.
      RETURN.
    ENDIF.

    DATA(lv_lh_name_xstr) = iv_zip_data+lv_name_off(lv_lh_fname_len).
    IF lv_lh_name_xstr <> is_entry-name_xstr.
      ev_message = 'File name mismatch between local header and central directory'.
      RETURN.
    ENDIF.

    IF lv_lh_method <> is_entry-method.
      ev_message = 'Compression method mismatch between local header and central directory'.
      RETURN.
    ENDIF.

    DATA(lv_has_data_desc) = xsdbool( ( ( is_entry-gp_flags DIV 8 ) MOD 2 ) = 1 ).
    IF lv_has_data_desc = abap_false.
      IF lv_lh_crc32 <> is_entry-crc32.
        ev_message = 'CRC32 mismatch between local header and central directory'.
        RETURN.
      ENDIF.

      IF lv_lh_comp_size <> is_entry-comp_size.
        ev_message = 'Compressed size mismatch between local header and central directory'.
        RETURN.
      ENDIF.

      IF lv_lh_uncomp_size <> is_entry-uncomp_size.
        ev_message = 'Uncompressed size mismatch between local header and central directory'.
        RETURN.
      ENDIF.
    ENDIF.

    DATA(lv_data_off) = lv_name_off + lv_lh_fname_len + lv_lh_extra_len.
    IF lv_data_off > lv_size.
      ev_message = 'Entry data offset exceeds ZIP size'.
      RETURN.
    ENDIF.

    IF is_entry-comp_size > 0
       AND lv_data_off + is_entry-comp_size > lv_size.
      ev_message = 'Entry compressed payload exceeds ZIP size'.
      RETURN.
    ENDIF.

    ev_ok = abap_true.
    ev_message = 'Entry consistency OK'.
  ENDMETHOD.

  METHOD verify_pdf_payload.
    ev_ok      = abap_false.
    ev_message = 'Unknown PDF validation error'.

    DATA(lv_size) = xstrlen( iv_pdf_data ).
    IF lv_size < 8.
      ev_message = 'PDF payload is too small'.
      RETURN.
    ENDIF.

    IF iv_pdf_data(4) <> c_pdf_magic.
      ev_message = 'Missing %PDF header'.
      RETURN.
    ENDIF.

    DATA(lv_tail_len) = COND i( WHEN lv_size > 1024 THEN 1024 ELSE lv_size ).
    DATA(lv_tail_off) = lv_size - lv_tail_len.
    DATA(lv_tail) = iv_pdf_data+lv_tail_off(lv_tail_len).

    DATA(lv_eof_upper_pos) = find_pattern(
      iv_haystack = lv_tail
      iv_needle   = CONV xstring( c_pdf_eof_upper ) ).
    DATA(lv_eof_lower_pos) = find_pattern(
      iv_haystack = lv_tail
      iv_needle   = CONV xstring( c_pdf_eof_lower ) ).

    IF lv_eof_upper_pos < 0 AND lv_eof_lower_pos < 0.
      ev_message = 'Missing %%EOF marker near end of file'.
      RETURN.
    ENDIF.

    ev_ok      = abap_true.
    ev_message = 'PDF payload integrity OK'.
  ENDMETHOD.

  METHOD find_pattern.
    rv_pos = -1.
    DATA(lv_hay_len) = xstrlen( iv_haystack ).
    DATA(lv_ned_len) = xstrlen( iv_needle ).

    IF lv_ned_len = 0 OR lv_ned_len > lv_hay_len.
      RETURN.
    ENDIF.

    "-- Search from the end to find the real EOCD signature and avoid
    "-- accidental matches inside compressed payload.
    DATA lv_i TYPE i.
    lv_i = lv_hay_len - lv_ned_len.

    WHILE lv_i >= 0.
      DATA(lv_slice) = iv_haystack+lv_i(lv_ned_len).
      IF lv_slice = iv_needle.
        rv_pos = lv_i.
        RETURN.
      ENDIF.
      lv_i = lv_i - 1.
    ENDWHILE.
  ENDMETHOD.

  METHOD read_uint16_le.
    "-- Read 2 bytes at offset as little-endian unsigned 16-bit
    DATA(lv_zip_data) = iv_xstring.
    DATA(lv_size) = xstrlen( lv_zip_data ).
    IF iv_offset + 2 > lv_size.
      rv = 0. RETURN.
    ENDIF.
    DATA(lv_bytes) = lv_zip_data+iv_offset(2).
    DATA lv_b0 TYPE x LENGTH 1.
    DATA lv_b1 TYPE x LENGTH 1.
    lv_b0 = lv_bytes(1).
    lv_b1 = lv_bytes+1(1).
    rv = ( CONV i( lv_b1 ) * 256 ) + CONV i( lv_b0 ).
  ENDMETHOD.

  METHOD read_uint32_le.
    "-- Read 4 bytes at offset as little-endian unsigned 32-bit
    DATA(lv_zip_data) = iv_xstring.
    DATA(lv_size) = xstrlen( lv_zip_data ).
    IF iv_offset + 4 > lv_size.
      rv = 0. RETURN.
    ENDIF.
    DATA(lv_bytes) = lv_zip_data+iv_offset(4).
    DATA lv_b0 TYPE x LENGTH 1.
    DATA lv_b1 TYPE x LENGTH 1.
    DATA lv_b2 TYPE x LENGTH 1.
    DATA lv_b3 TYPE x LENGTH 1.
    lv_b0 = lv_bytes(1).
    lv_b1 = lv_bytes+1(1).
    lv_b2 = lv_bytes+2(1).
    lv_b3 = lv_bytes+3(1).

    DATA(lv_b0_i8) = CONV int8( CONV i( lv_b0 ) ).
    DATA(lv_b1_i8) = CONV int8( CONV i( lv_b1 ) ).
    DATA(lv_b2_i8) = CONV int8( CONV i( lv_b2 ) ).
    DATA(lv_b3_i8) = CONV int8( CONV i( lv_b3 ) ).

    DATA(lv_u32_i8) =   ( lv_b3_i8 * CONV int8( 16777216 ) )
                      + ( lv_b2_i8 * CONV int8( 65536    ) )
                      + ( lv_b1_i8 * CONV int8( 256      ) )
                      +   lv_b0_i8.

    " Clamp to i range to avoid arithmetic overflow when ZIP UINT32 > INT4 max
    IF lv_u32_i8 > CONV int8( 2147483647 ).
      rv = 2147483647.
    ELSE.
      rv = CONV i( lv_u32_i8 ).
    ENDIF.
  ENDMETHOD.

ENDCLASS.

"--------------------------------------------------------------------
" Local class: Main orchestrator
"--------------------------------------------------------------------
CLASS lcl_main DEFINITION FINAL.
  PUBLIC SECTION.

    "! Entry point called from START-OF-SELECTION.
    CLASS-METHODS run
      IMPORTING iv_is_local   TYPE abap_bool
                iv_path       TYPE string
                it_file_sel   TYPE ty_t_file_sel
                iv_strict     TYPE abap_bool.

  PRIVATE SECTION.

    "! Processes a list of ZIP file paths and returns check results.
    CLASS-METHODS process_files
      IMPORTING iv_is_local  TYPE abap_bool
                iv_strict    TYPE abap_bool
                it_files     TYPE string_table
      RETURNING VALUE(rt_res) TYPE ty_t_check_result.

ENDCLASS.

CLASS lcl_main IMPLEMENTATION.

  METHOD run.
    "-- Authorization check first
    TRY.
        lcl_auth_checker=>check_file_auth( ).
      CATCH lcx_no_authority INTO DATA(lx_auth).
        MESSAGE 'No authorization for file operations (S_DATASET ACTVT=33)' TYPE 'E'.
        RETURN.
    ENDTRY.

    DATA lt_files TYPE string_table.
    DATA lt_files_filtered TYPE string_table.

    "-- Collect ZIP file list
    TRY.
        IF iv_is_local = abap_true.
          lcl_file_utils=>get_local_zip_files(
            EXPORTING iv_path  = iv_path
            IMPORTING et_files = lt_files ).
        ELSE.
          lcl_file_utils=>get_server_zip_files(
            EXPORTING iv_path  = iv_path
            IMPORTING et_files = lt_files ).
        ENDIF.
      CATCH cx_parameter_invalid INTO DATA(lx_dir).
        MESSAGE lx_dir->get_text( ) TYPE 'E'.
        RETURN.
    ENDTRY.

    IF lt_files IS INITIAL.
      MESSAGE |No .zip files found in: { iv_path }| TYPE 'I'.
      RETURN.
    ENDIF.

    IF it_file_sel IS NOT INITIAL.
      LOOP AT lt_files INTO DATA(lv_full_path).
        DATA(lv_name_only) = lcl_file_utils=>get_filename_from_path( lv_full_path ).
        IF lv_name_only IN it_file_sel.
          APPEND lv_full_path TO lt_files_filtered.
        ENDIF.
      ENDLOOP.
    ELSE.
      lt_files_filtered = lt_files.
    ENDIF.

    IF lt_files_filtered IS INITIAL.
      MESSAGE 'No ZIP files match the select-options filter' TYPE 'I'.
      RETURN.
    ENDIF.

    "-- Process each file
    DATA(lt_results) = process_files(
      iv_is_local = iv_is_local
      iv_strict   = iv_strict
      it_files    = lt_files_filtered ).

    "-- Display ALV
    lcl_alv_output=>display( it_results = lt_results ).
  ENDMETHOD.

  METHOD process_files.
    DATA lt_res TYPE ty_t_check_result.

    LOOP AT it_files INTO DATA(lv_filepath).
      DATA(lv_fname) = lcl_file_utils=>get_filename_from_path( lv_filepath ).
      DATA ls_result TYPE ty_check_result.

      ls_result-file_name = lv_fname.
      ls_result-full_path = lv_filepath.

      TRY.
          DATA lv_xstr TYPE xstring.

          IF iv_is_local = abap_true.
            lcl_file_utils=>read_local_file(
              EXPORTING iv_filepath = lv_filepath
              IMPORTING ev_xstring  = lv_xstr ).
          ELSE.
            lcl_file_utils=>read_server_file(
              EXPORTING iv_filepath = lv_filepath
              IMPORTING ev_xstring  = lv_xstr ).
          ENDIF.

          "-- Verify ZIP integrity
          DATA lv_ok  TYPE abap_bool.
          DATA lv_msg TYPE string.
          DATA lv_warn TYPE string.
          DATA lv_eocd_pos TYPE i.
          DATA lv_zip_size TYPE i.
          DATA lv_trailing_b TYPE i.
          lcl_zip_verifier=>verify_zip_integrity(
            EXPORTING iv_xstring  = lv_xstr
                      iv_strict   = iv_strict
            IMPORTING ev_ok       = lv_ok
                      ev_message  = lv_msg
                      ev_warning  = lv_warn
                      ev_eocd_pos = lv_eocd_pos
                      ev_zip_size = lv_zip_size
                      ev_trailing_b = lv_trailing_b ).

          ls_result-check_result = COND #( WHEN lv_ok = abap_true THEN 'OK' ELSE 'KO' ).
          ls_result-msg_error    = COND #( WHEN lv_ok = abap_true THEN '' ELSE lv_msg ).
          ls_result-msg_warn     = COND #( WHEN lv_ok = abap_true THEN lv_warn ELSE '' ).
          ls_result-icon         = COND #( WHEN lv_ok = abap_true THEN '@5B@' ELSE '@5C@' ).
          ls_result-eocd_pos     = lv_eocd_pos.
          ls_result-zip_size     = lv_zip_size.
          ls_result-trailing_b   = lv_trailing_b.

        CATCH cx_parameter_invalid INTO DATA(lx_par).
          ls_result-check_result = 'KO'.
          ls_result-msg_error    = lx_par->get_text( ).
          ls_result-msg_warn     = ''.
          ls_result-icon         = '@5C@'.
        CATCH cx_root INTO DATA(lx_root).
          ls_result-check_result = 'KO'.
          ls_result-msg_error    = lx_root->get_text( ).
          ls_result-msg_warn     = ''.
          ls_result-icon         = '@5C@'.
      ENDTRY.

      APPEND ls_result TO lt_res.
    ENDLOOP.

    rt_res = lt_res.
  ENDMETHOD.

ENDCLASS.


"--------------------------------------------------------------------
" AT SELECTION-SCREEN: field-level validations
"--------------------------------------------------------------------
AT SELECTION-SCREEN OUTPUT.
  LOOP AT SCREEN.
    IF screen-group1 = 'LOC'.
      screen-input = COND #( WHEN p_local = abap_true THEN 1 ELSE 0 ).
      MODIFY SCREEN.
    ELSEIF screen-group1 = 'SRV'.
      screen-input = COND #( WHEN p_servr = abap_true THEN 1 ELSE 0 ).
      MODIFY SCREEN.
    ENDIF.
  ENDLOOP.

AT SELECTION-SCREEN.
  IF p_local = abap_true AND p_ldir IS INITIAL.
    MESSAGE 'Indicare la directory locale da controllare' TYPE 'E'.
  ENDIF.

  IF p_servr = abap_true AND p_sdir IS INITIAL.
    MESSAGE 'Indicare la directory sul server SAP da controllare' TYPE 'E'.
  ENDIF.

"--------------------------------------------------------------------
" START-OF-SELECTION
"--------------------------------------------------------------------
START-OF-SELECTION.
  DATA lv_run_local TYPE abap_bool.
  DATA lv_run_path  TYPE string.
  DATA lv_strict    TYPE abap_bool.

  lv_run_local = COND #( WHEN p_local = abap_true THEN abap_true ELSE abap_false ).
  lv_run_path  = COND #( WHEN p_local = abap_true THEN CONV string( p_ldir ) ELSE CONV string( p_sdir ) ).
  lv_strict    = COND #( WHEN p_strict = abap_true THEN abap_true ELSE abap_false ).

  IF lv_run_path NS '/' AND lv_run_path NS '\'.
    lv_run_path = |{ lv_run_path }/|.
  ELSE.
    DATA(lv_last_char) = substring( val = lv_run_path off = strlen( lv_run_path ) - 1 len = 1 ).
    IF lv_last_char <> '/' AND lv_last_char <> '\'.
      lv_run_path = |{ lv_run_path }/|.
    ENDIF.
  ENDIF.

  lcl_main=>run(
    iv_is_local = lv_run_local
    iv_path     = lv_run_path
    it_file_sel = s_file[]
    iv_strict   = lv_strict ).
