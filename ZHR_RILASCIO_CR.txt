  METHOD user_command.
    IF iv_ucomm = '&IC1' AND cs_selfield-fieldname = 'MESSAGE'.
      display_release_log( cs_selfield-tabindex ).
      RETURN.
    ENDIF.

    IF iv_ucomm = '&ETA' OR
       ( iv_ucomm = '&IC1' AND cs_selfield-fieldname = 'BUTTON' ).
      process_transports( ).
      cs_selfield-refresh = abap_true.

  METHOD display_release_log.
    ASSIGN mt_alv[ iv_row_index ] TO FIELD-SYMBOL(<ls_alv>).
    IF sy-subrc <> 0.
      RETURN.
    ENDIF.

    IF <ls_alv>-release_error = abap_true.
      display_release_error(
        iv_trkorr  = <ls_alv>-trkorr
        iv_message = <ls_alv>-release_log ).
      RETURN.
    ENDIF.

    DATA lt_trkorrs TYPE ctslg_trkorrs.
    DATA lv_title TYPE trwbo_title.

    APPEND <ls_alv>-trkorr TO lt_trkorrs.
    lv_title = |Log rilascio CR { <ls_alv>-trkorr }|.

    CALL FUNCTION 'TR_LOG_OVERVIEW_REQUESTS'
      EXPORTING
        it_trkorrs             = lt_trkorrs
        iv_dir_type            = 'T'
        iv_title               = lv_title
      EXCEPTIONS
        e_wrong_call           = 1
        OTHERS                  = 2.

    IF sy-subrc <> 0.
      MESSAGE 'Impossibile visualizzare il log standard della richiesta' TYPE 'S' DISPLAY LIKE 'E'.
    ENDIF.
  ENDMETHOD.

  METHOD display_release_error.
    DATA lt_error_log TYPE tt_error_log.
    DATA lt_log_lines TYPE string_table.
    DATA lt_fieldcat TYPE slis_t_fieldcat_alv.
    DATA lv_log_line TYPE string.

    lt_fieldcat = VALUE #(
      ( fieldname = 'LOG_TEXT' seltext_m = 'Dettaglio errore rilascio' outputlen = 110 ) ).
    SPLIT iv_message AT cl_abap_char_utilities=>newline INTO TABLE lt_log_lines.

    LOOP AT lt_log_lines INTO lv_log_line.
      DO.
        IF strlen( lv_log_line ) <= 255.
          APPEND VALUE #( log_text = lv_log_line ) TO lt_error_log.
          EXIT.
        ENDIF.

        APPEND VALUE #( log_text = lv_log_line(255) ) TO lt_error_log.
        lv_log_line = lv_log_line+255.
      ENDDO.
    ENDLOOP.

    CALL FUNCTION 'REUSE_ALV_POPUP_TO_SELECT'
      EXPORTING
        i_tabname              = 'LT_ERROR_LOG'
        i_screen_start_column  = 5
        i_screen_start_line    = 3
        i_screen_end_column    = 120
        i_screen_end_line      = 20
        it_fieldcat            = lt_fieldcat
      TABLES
        t_outtab               = lt_error_log
      EXCEPTIONS
        program_error          = 1
        OTHERS                 = 2.
  ENDMETHOD.

  METHOD append_release_log.

      WHEN cv_release_log IS INITIAL THEN iv_message
      ELSE |{ cv_release_log }{ cl_abap_char_utilities=>newline }{ iv_message }| ).
  ENDMETHOD.

  METHOD process_transports.
    DATA(lv_success) = 0.
    DATA(lv_errors)  = 0.

    IF authorize_check( ) = abap_false.
      MESSAGE 'Autorizzazione mancante per il rilascio (S_TRANSPRT)' TYPE 'E'.
      RETURN.
    ENDIF.

    LOOP AT mt_alv ASSIGNING FIELD-SYMBOL(<ls_alv>) WHERE check = 'X'.
      DATA(lv_row_index) = sy-tabix.
      CLEAR <ls_alv>-release_log.
      CLEAR <ls_alv>-release_error.

      " 1. Identificazione Sottotask (Tasks) legati alla CR
      SELECT trkorr
        FROM e070
        WHERE strkorr = @<ls_alv>-trkorr
          AND trstatus = 'D' " Solo quelli ancora modificabili
        INTO TABLE @DATA(lt_subtasks).

      IF p_test = abap_true.
        <ls_alv>-message = |[SIMULAZIONE] Rilascio { lines( lt_subtasks ) } task e CR { <ls_alv>-trkorr } OK|.
        append_release_log(
          EXPORTING iv_message = <ls_alv>-message
          CHANGING  cv_release_log = <ls_alv>-release_log ).
        lv_success = lv_success + 1.
        CONTINUE.
      ENDIF.

      " 2. Rilascio preventivo dei Sottotask
      DATA(lv_subtask_error) = abap_false.
      LOOP AT lt_subtasks INTO DATA(ls_subtask).
        CALL FUNCTION 'TR_RELEASE_REQUEST'
          EXPORTING
            iv_trkorr          = ls_subtask-trkorr
            iv_dialog          = abap_false
            iv_success_message = abap_false
          EXCEPTIONS
            OTHERS             = 1.
        IF sy-subrc <> 0.
          lv_subtask_error = abap_true.
          MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                  WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                  INTO DATA(lv_subtask_message).
          DATA(lv_subtask_error_log) = |Task { ls_subtask-trkorr }: { lv_subtask_message }|.
          append_release_log(
            EXPORTING iv_message = lv_subtask_error_log
            CHANGING  cv_release_log = <ls_alv>-release_log ).
          EXIT. " Se un sottotask fallisce, la CR principale fallir# sicuramente
        ENDIF.
        DATA(lv_subtask_success_log) = |Task { ls_subtask-trkorr }: rilasciata con successo|.
        append_release_log(
          EXPORTING iv_message = lv_subtask_success_log
          CHANGING  cv_release_log = <ls_alv>-release_log ).
      ENDLOOP.

      IF lv_subtask_error = abap_true.
        lv_errors = lv_errors + 1.
        <ls_alv>-release_error = abap_true.
        <ls_alv>-message = |Errore durante il rilascio di uno dei sottotask|.
        append_release_log(
          EXPORTING iv_message = <ls_alv>-message
          CHANGING  cv_release_log = <ls_alv>-release_log ).
        display_release_error(
          iv_trkorr  = <ls_alv>-trkorr
          iv_message = <ls_alv>-release_log ).
        CONTINUE.
      ENDIF.

      " 3. Rilascio della Transport Request Principale (CR)
      CALL FUNCTION 'TR_RELEASE_REQUEST'
        EXPORTING
          iv_trkorr                  = <ls_alv>-trkorr
          iv_dialog                  = abap_false
          iv_success_message         = abap_false
        EXCEPTIONS
          cts_initialization_failure = 1
          enqueue_failed             = 2
          no_authorization           = 3
          invalid_request            = 4
          request_already_released   = 5
          OTHERS                     = 6.

      IF sy-subrc = 0.
        <ls_alv>-message = 'Rilascio (Tasks + CR) completato con successo'.
        DATA(lv_cr_success_log) = |CR { <ls_alv>-trkorr }: rilasciata con successo|.
        append_release_log(
          EXPORTING iv_message = lv_cr_success_log
          CHANGING  cv_release_log = <ls_alv>-release_log ).
        lv_success = lv_success + 1.
      ELSE.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                INTO <ls_alv>-message.
        <ls_alv>-release_error = abap_true.
        DATA(lv_cr_error_log) = |CR { <ls_alv>-trkorr }: { <ls_alv>-message }|.
        append_release_log(
          EXPORTING iv_message = lv_cr_error_log
          CHANGING  cv_release_log = <ls_alv>-release_log ).
        display_release_error(
          iv_trkorr  = <ls_alv>-trkorr
          iv_message = <ls_alv>-release_log ).
        lv_errors = lv_errors + 1.
      ENDIF.
    ENDLOOP.

    " Log Finale in Pop-up
    IF lv_success > 0 OR lv_errors > 0.
      DATA(lv_log) = |Elaborazione terminata.\n\n| &&
                     |Successi: { lv_success }\n| &&
                     |Errori: { lv_errors }|.
      MESSAGE lv_log TYPE 'I'.
    ELSE.
      MESSAGE 'Nessuna riga selezionata tramite checkbox' TYPE 'S' DISPLAY LIKE 'E'.
    ENDIF.
  ENDMETHOD.

ENDCLASS.

FORM f4_trkorr USING iv_dynprofield TYPE dynfnam.
  DATA lt_trkorr_f4 TYPE tt_trkorr_f4.

  SELECT main~trkorr,
         main~as4user,
         text~as4text
    FROM e070 AS main
    LEFT OUTER JOIN e07t AS text
      ON main~trkorr = text~trkorr
     AND text~langu  = @sy-langu
    WHERE main~trstatus = 'D'
      AND ( main~as4user = @sy-uname
         OR EXISTS ( SELECT sub~trkorr
                       FROM e070 AS sub
                      WHERE sub~strkorr = main~trkorr
                        AND sub~as4user = @sy-uname ) )
      AND ( main~korrdev = 'CUST' )
      and TRFUNCTION = 'W'
*    ORDER BY trkorr
    INTO TABLE @lt_trkorr_f4.

  IF lt_trkorr_f4 IS INITIAL.
    MESSAGE 'Nessuna CR modificabile trovata per l''utente indicato' TYPE 'S' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield    = 'TRKORR'
      dynpprog    = sy-repid
      dynpnr      = sy-dynnr
      dynprofield = iv_dynprofield
      value_org   = 'S'
    TABLES
      value_tab   = lt_trkorr_f4
    EXCEPTIONS
      parameter_error = 1
      no_values_found = 2
      OTHERS          = 3.

  IF sy-subrc <> 0 AND sy-subrc <> 2.
    MESSAGE 'Impossibile visualizzare il matchcode delle CR' TYPE 'S' DISPLAY LIKE 'E'.
  ENDIF.
ENDFORM.
