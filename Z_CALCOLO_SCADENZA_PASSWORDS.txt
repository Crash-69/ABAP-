*&---------------------------------------------------------------------*
*& Report Z_CALCOLO_SCADENZA_PASSWORDS
*&---------------------------------------------------------------------*
*& Auth.: Crash_AI 2026
*& Descrizione: Calcola la data di scadenza della password e i giorni
*&              rimanenti per una selezione di utenti.
*&---------------------------------------------------------------------*
REPORT z_calcolo_scadenza_passwords.

TABLES: usr02.

* Struct per l'output della tabella ALV
TYPES: BEGIN OF ty_output,
         bname     TYPE usr02-bname,
         pwdchdate TYPE usr02-pwdchgdate,
         exp_date  TYPE d,
         days_left TYPE i,
         status    TYPE string,
       END OF ty_output.

DATA: lt_usr02  TYPE TABLE OF usr02,
      ls_usr02  TYPE usr02,
      lt_output TYPE TABLE OF ty_output,
      ls_output TYPE ty_output,
      lv_val    TYPE c LENGTH 60,
      lv_days   TYPE i,
      lo_alv    TYPE REF TO cl_salv_table,
      lo_cols   TYPE REF TO cl_salv_columns_table,
      lo_col    TYPE REF TO cl_salv_column_table,
      lo_functions TYPE REF TO cl_salv_functions_list.

*----------------------------------------------------------------------*
* SCHERMATA DI SELEZIONE
*----------------------------------------------------------------------*
SELECTION-SCREEN BEGIN OF BLOCK b1 WITH FRAME TITLE TEXT-001.
  SELECT-OPTIONS: s_user FOR usr02-bname MATCHCODE OBJECT user_comp.
SELECTION-SCREEN END OF BLOCK b1.

INITIALIZATION.
**  TEXT-001 = 'Selezione Utenti'.

*----------------------------------------------------------------------*
* START-OF-SELECTION
*----------------------------------------------------------------------*
START-OF-SELECTION.

  " 1. Legge il parametro del profilo di sistema per la validità della password
  CALL 'C_SAPGPARAM' ID 'NAME'  FIELD 'login/password_expiration_time'
                     ID 'VALUE' FIELD lv_val.

  lv_days = lv_val.

  " 2. Seleziona i dati degli utenti inseriti
  SELECT * FROM usr02
    INTO TABLE lt_usr02
    WHERE bname IN s_user.

  IF lt_usr02 IS INITIAL.
    MESSAGE 'Nessun utente trovato per i criteri inseriti.' TYPE 'I' DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.

  " 3. Elaborazione dei dati e calcolo delle scadenze
  LOOP AT lt_usr02 INTO ls_usr02.
    CLEAR ls_output.
    ls_output-bname     = ls_usr02-bname.
    ls_output-pwdchdate = ls_usr02-pwdchgdate.

    " Se l'utente non ha mai cambiato password o non è valorizzata, usa la data di creazione
    IF ls_output-pwdchdate IS INITIAL OR ls_output-pwdchdate = '00000000'.
      " In subordine usiamo la data dell'ultimo login o lasciamo vuoto
      ls_output-status = 'Mai modificata / Stato iniziale'.
    ENDIF.

    " Calcolo se il parametro di sistema prevede una scadenza (> 0 giorni)
    IF lv_days > 0.
      IF ls_output-pwdchdate IS NOT INITIAL AND ls_output-pwdchdate <> '00000000'.

        " Data di scadenza = Data ultimo cambio + Giorni di validità parametro
        ls_output-exp_date = ls_output-pwdchdate + lv_days.

        " Giorni rimanenti = Data di Scadenza - Data Odierna
        ls_output-days_left = ls_output-exp_date - sy-datum.

        IF ls_output-days_left < 0.
          ls_output-status = 'SCADUTA'.
        ELSE.
          ls_output-status = 'ATTIVA'.
        ENDIF.
      ENDIF.
    ELSE.
      ls_output-status = 'La password non scade mai (Parametro di sistema a 0)'.
    ENDIF.

    APPEND ls_output TO lt_output.
  ENDLOOP.

*----------------------------------------------------------------------*
* END-OF-SELECTION (Generazione Output Griglia ALV)
*----------------------------------------------------------------------*
END-OF-SELECTION.

  TRY.
      " Creazione dell'istanza ALV in Object Oriented
      cl_salv_table=>factory(
        IMPORTING
          r_salv_table = lo_alv
        CHANGING
          t_table      = lt_output ).

      " Attivazione delle funzioni standard ALV (Filtri, Ordinamento, Export Excel)
      lo_functions = lo_alv->get_functions( ).
      lo_functions->set_all( abap_true ).

      " Personalizzazione delle intestazioni di colonna
      lo_cols = lo_alv->get_columns( ).
      lo_cols->set_optimize( abap_true ).

      lo_col ?= lo_cols->get_column( 'BNAME' ).
      lo_col->set_short_text( 'Utente' ).
      lo_col->set_medium_text( 'Nome Utente' ).

      lo_col ?= lo_cols->get_column( 'PWDCHDATE' ).
      lo_col->set_short_text( 'Ult.Cambio' ).
      lo_col->set_medium_text( 'Data Ultimo Cambio' ).

      lo_col ?= lo_cols->get_column( 'EXP_DATE' ).
      lo_col->set_short_text( 'Scadenza' ).
      lo_col->set_medium_text( 'Data Scadenza Pwd' ).

      lo_col ?= lo_cols->get_column( 'DAYS_LEFT' ).
      lo_col->set_short_text( 'Gg.Residui' ).
      lo_col->set_medium_text( 'Giorni Rimanenti' ).

      lo_col ?= lo_cols->get_column( 'STATUS' ).
      lo_col->set_short_text( 'Stato' ).
      lo_col->set_medium_text( 'Stato Password' ).

      " Mostra la tabella a schermo
      lo_alv->display( ).

    CATCH cx_salv_msg.
      MESSAGE 'Errore nella generazione del report ALV.' TYPE 'E'.
  ENDTRY.
