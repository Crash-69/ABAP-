  METHOD user_command.
    IF iv_ucomm = '&IC1' AND cs_selfield-fieldname = 'MESSAGE'.
      display_import_log( cs_selfield-tabindex ).
      RETURN.
    ENDIF.

    IF iv_ucomm = '&ETA' OR
       ( iv_ucomm = '&IC1' AND cs_selfield-fieldname = 'BUTTON' ).
      process_imports( ).
      cs_selfield-refresh = abap_true.
    ENDIF.
  ENDMETHOD.

  METHOD display_import_log.
    ASSIGN mt_alv[ iv_row_index ] TO FIELD-SYMBOL(<ls_alv>).
    IF sy-subrc <> 0.
      RETURN.
    ENDIF.

    DATA lt_trkorrs TYPE ctslg_trkorrs.
    DATA lv_title TYPE trwbo_title.

    APPEND <ls_alv>-trkorr TO lt_trkorrs.
    lv_title = |Log import CR { <ls_alv>-trkorr }|.

    CALL FUNCTION 'TR_LOG_OVERVIEW_REQUESTS'
      EXPORTING
        it_trkorrs             = lt_trkorrs
        iv_dir_type            = 'T'
        iv_title               = lv_title
      EXCEPTIONS
        e_wrong_call           = 1
        OTHERS                 = 2.

    IF sy-subrc <> 0.
      MESSAGE 'Impossibile visualizzare il log standard della richiesta' TYPE 'S' DISPLAY LIKE 'E'.
    ENDIF.
  ENDMETHOD.

  METHOD process_imports.
    DATA(lv_success) = 0.
    DATA(lv_errors)  = 0.

    LOOP AT mt_alv ASSIGNING FIELD-SYMBOL(<ls_alv>) WHERE check = 'X'.
      CLEAR <ls_alv>-import_log.

      IF p_test = abap_true.
        <ls_alv>-message = |[TEST] Simulazione Import in { p_dest2 } OK|.
        lv_success = lv_success + 1.
        CONTINUE.
      ENDIF.

      " --- LOGICA DI IMPORT (Simile a STMS) ---
      DATA: lv_tp_rc type TRRETCODE.
      DATA: lt_trkorr TYPE tmsbuffer-trkorr.

      lt_trkorr = <ls_alv>-trkorr(10).

      CALL FUNCTION 'TMS_MGR_IMPORT_TR_REQUEST'
        EXPORTING
          iv_system             = p_dest2
          iv_request            = lt_trkorr
          iv_client             = sy-mandt
          IV_IMPORT_AGAIN       = abap_true
          iv_ignore_originality = abap_true
          iv_ignore_repairs     = abap_true
          iv_ignore_transtype   = abap_true
          iv_ignore_predec      = abap_true
          iv_ignore_cvers       = abap_true
          iv_monitor            = abap_true
          iv_verbose            = abap_true
        IMPORTING
          ev_tp_ret_code        = lv_tp_rc
        EXCEPTIONS
          OTHERS                = 1.

      IF sy-subrc = 0 AND lv_tp_rc <= 4.
        <ls_alv>-message = |Import eseguito con RC={ lv_tp_rc }|.
        lv_success = lv_success + 1.
      ELSE.
        lv_errors = lv_errors + 1.
        <ls_alv>-message = |Errore Import. RC={ lv_tp_rc }. Verifica STMS.|.
      ENDIF.
    ENDLOOP.

    IF lv_success > 0 OR lv_errors > 0.
      MESSAGE |Import completato. Successi: { lv_success }, Errori: { lv_errors }| TYPE 'I'.
    ELSE.
      MESSAGE 'Selezionare almeno una CR rilasciata' TYPE 'S' DISPLAY LIKE 'E'.
    ENDIF.
  ENDMETHOD.

FORM f4_trkorr USING iv_dynprofield TYPE dynfnam.
  DATA lt_trkorr_f4 TYPE tt_trkorr_f4.

  SELECT main~trkorr,
         main~as4user,
         text~as4text
    FROM e070 AS main
    LEFT OUTER JOIN e07t AS text
      ON main~trkorr = text~trkorr
     AND text~langu  = @sy-langu
    WHERE main~trstatus = 'R'
      AND ( main~as4user = @sy-uname
         OR EXISTS ( SELECT sub~trkorr
                       FROM e070 AS sub
                      WHERE sub~strkorr = main~trkorr
                        AND sub~as4user = @sy-uname ) )
      and TRFUNCTION = 'W'
      AND ( main~korrdev = 'CUST')
*    ORDER BY trkorr
    INTO TABLE @lt_trkorr_f4.

  IF lt_trkorr_f4 IS INITIAL.
    MESSAGE 'Nessuna CR modificabile trovata per l''utente indicato' TYPE 'S' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield    = 'TRKORR'
      dynpprog    = sy-repid
      dynpnr      = sy-dynnr
      dynprofield = iv_dynprofield
      value_org   = 'S'
    TABLES
      value_tab   = lt_trkorr_f4
    EXCEPTIONS
      parameter_error = 1
      no_values_found = 2
      OTHERS          = 3.

  IF sy-subrc <> 0 AND sy-subrc <> 2.
    MESSAGE 'Impossibile visualizzare il matchcode delle CR' TYPE 'S' DISPLAY LIKE 'E'.
  ENDIF.
ENDFORM.
